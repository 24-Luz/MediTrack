public class Medico {

    // Constructores
    
    public Medico (String id_medico, String nombre, String especialidad) {
        this.id_medico = id_medico;
        this.nombre = nombre;
        this.especialidad = especialidad; 
    }

        this.id_medico = id_medico;
        this.nombre = nombre;
        this.especialidad = "General"; // especialidad por defecto
        
    }
    
    // Metodos
    public void mostrar_informacion (){
        System.out.println("Medico - Especialista");
        System.out.println("ID Medico: " + id_medico);

        System.out.println("Especialidad: " + especialidad);
    }
    
    public void aceptar_cita(){
        System.out.println("El medico " + nombre + " ha aceptado la cita.");
    }
    
    public void rechazar_cita(){
        System.out.println("El medico " + nombre + " ha rechazado la cita.");
    }
